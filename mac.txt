Option Explicit

' =====================================================================
'  Platform onboarding deck builder
'  Run "BuildOnboardingDeck". It creates a NEW presentation (14 slides)
'  with Morph transitions and entrance animations. Your open files are
'  not touched.
'  Palette: white base, dark blue, light blue, light pink, yellow.
' =====================================================================

Private pres As Presentation
Private sld As Slide
Private firstAnim As Boolean

Private cNavy As Long, cSky As Long, cSkyStrong As Long, cPink As Long, cPinkFg As Long
Private cYellow As Long, cFg As Long, cMuted As Long, cWhite As Long
Private cLine As Long, cOnNavy As Long
Private DOT As String

Private Const TOTAL As Integer = 14
Private Const FONT_HEAD As String = "Segoe UI Semibold"
Private Const FONT_BODY As String = "Segoe UI"
Private Const FONT_BLACK As String = "Segoe UI Black"
Private Const MORPH As Long = 3953   ' ppEffectMorphByObject (PowerPoint 2019 / Microsoft 365)

Public Sub BuildOnboardingDeck()
    InitStyle
    Set pres = Presentations.Add(msoTrue)
    pres.PageSetup.SlideWidth = 960
    pres.PageSetup.SlideHeight = 540

    S01_Title
    S02_Architecture
    S03_Agenda
    S04_WhoOwns
    S05_OwnershipMap
    S06_Setup
    S07_Contract
    S08_QualityGates
    S09_Releases
    S10_WhenThingsBreak
    S11_Production
    S12_Help
    S13_FirstWeek
    S14_Close

    MsgBox "Your onboarding deck is ready." & vbCr & vbCr & _
           "Press F5 in PowerPoint to play it with animations." & vbCr & _
           "Replace the [placeholders] with your real details.", vbInformation, "Onboarding deck"
End Sub

' ---------------------------------------------------------------------
'  Style
' ---------------------------------------------------------------------
Private Sub InitStyle()
    cNavy = RGB(19, 41, 107)
    cSky = RGB(230, 240, 252)
    cSkyStrong = RGB(185, 211, 243)
    cPink = RGB(253, 228, 236)
    cPinkFg = RGB(138, 42, 79)
    cYellow = RGB(255, 241, 184)
    cFg = RGB(14, 26, 58)
    cMuted = RGB(79, 91, 117)
    cWhite = RGB(255, 255, 255)
    cLine = RGB(227, 232, 242)
    cOnNavy = RGB(188, 211, 242)
    DOT = "  " & ChrW(183) & "  "
End Sub

' ---------------------------------------------------------------------
'  Slide scaffolding
' ---------------------------------------------------------------------
Private Sub NewSlide(ByVal idx As Integer, ByVal notes As String)
    Set sld = pres.Slides.Add(idx, ppLayoutBlank)
    sld.FollowMasterBackground = msoFalse
    sld.Background.Fill.Solid
    sld.Background.Fill.ForeColor.RGB = cWhite
    firstAnim = True

    Blobs idx
    Progress idx
    If idx > 1 And idx < TOTAL Then Footer idx

    On Error Resume Next
    sld.SlideShowTransition.EntryEffect = MORPH
    If Err.Number <> 0 Then Err.Clear: sld.SlideShowTransition.EntryEffect = ppEffectFadeSmoothly
    sld.SlideShowTransition.Duration = 1.1
    sld.NotesPage.Shapes.Placeholders(2).TextFrame.TextRange.Text = notes
    Err.Clear
    On Error GoTo 0
End Sub

' Soft pink and yellow circles. Same names on every slide, so Morph glides them between slides.

    px = Array(620, 700, 760, 740, 690, 760, 720, 700, 760, 740, 700, 720, 760, 560)
    py = Array(-200, -230, -250, -240, -260, -230, -250, -240, -260, -230, -240, -250, -230, -180)
    ps = Array(460, 340, 300, 320, 300, 320, 300, 320, 300, 320, 300, 320, 320, 520)
    yx = Array(-140, -100, -80, -120, -90, -110, -80, -100, -90, -120, -100, -80, -110, -160)
    yy = Array(380, 420, 440, 430, 450, 420, 440, 430, 450, 420, 440, 430, 440, 340)
    ys = Array(320, 240, 220, 240, 220, 240, 220, 240, 220, 240, 220, 240, 240, 380)
    i = idx - 1

    Box yx(i), yy(i), ys(i), ys(i), cYellow, 0, -1, "!!blobYellow", msoShapeOval

' Thin progress bar along the bottom edge; Morph grows it slide to slide.
Private Sub Progress(ByVal idx As Integer)
    Box 0, 536, 960, 4, cSky, 0, -1, "!!track", msoShapeRectangle
    Box 0, 536, 960 * idx / TOTAL, 4, cNavy, 0, -1, "!!progress", msoShapeRectangle
End Sub

Private Sub Footer(ByVal idx As Integer)
    Txt 56, 506, 500, 16, "[App name]" & DOT & "Platform onboarding", 9, cMuted
    Txt 804, 506, 100, 16, Format(idx, "00") & " / " & Format(TOTAL, "00"), 9, cMuted, FONT_BODY, ppAlignRight

' ---------------------------------------------------------------------
'  Building blocks
' ---------------------------------------------------------------------
Private Function Box(ByVal x As Single, ByVal y As Single, ByVal w As Single, ByVal h As Single, _
                     ByVal fillC As Long, Optional ByVal radius As Single = 0.12, _
                     Optional ByVal lineC As Long = -1, Optional ByVal nm As String = "", _
                     Optional ByVal shp As Long = 5) As Shape
    Dim s As Shape
    If radius = 0 And shp = msoShapeRoundedRectangle Then shp = msoShapeRectangle
    Set s = sld.Shapes.AddShape(shp, x, y, w, h)
    If shp = msoShapeRoundedRectangle Or shp = msoShapeRound2SameRectangle Then s.Adjustments(1) = radius
    s.Fill.Solid
    s.Fill.ForeColor.RGB = fillC
    If lineC = -1 Then
        s.Line.Visible = msoFalse
    Else
        s.Line.Visible = msoTrue
        s.Line.ForeColor.RGB = lineC
        s.Line.Weight = 1
    End If
    s.Shadow.Visible = msoFalse
    If nm <> "" Then s.Name = nm
    Set Box = s
End Function

Private Function Txt(ByVal x As Single, ByVal y As Single, ByVal w As Single, ByVal h As Single, _
                     ByVal t As String, ByVal size As Single, ByVal col As Long, _
                     Optional ByVal fontName As String = FONT_BODY, Optional ByVal align As Long = 1) As Shape
    Dim s As Shape
    Set s = sld.Shapes.AddTextbox(msoTextOrientationHorizontal, x, y, w, h)
    With s.TextFrame
        .WordWrap = msoTrue
        .MarginLeft = 0: .MarginRight = 0: .MarginTop = 0: .MarginBottom = 0
        .TextRange.Text = t
        .TextRange.Font.Name = fontName
        .TextRange.Font.Size = size
        .TextRange.Font.Color.RGB = col
        .TextRange.ParagraphFormat.Alignment = align
        .AutoSize = ppAutoSizeNone
    End With
    Set Txt = s
End Function

Private Sub StylePara(ByVal s As Shape, ByVal i As Integer, ByVal size As Single, ByVal col As Long, _
                      ByVal fontName As String, Optional ByVal spacing As Single = 0)
    With s.TextFrame.TextRange.Paragraphs(i).Font
        .Size = size
        .Color.RGB = col
        .Name = fontName
    End With
    If spacing <> 0 Then
        On Error Resume Next
        s.TextFrame2.TextRange.Paragraphs(i).Font.Spacing = spacing
        On Error GoTo 0
    End If
End Sub

Private Sub BoxText(ByVal s As Shape, ByVal t As String, ByVal size As Single, ByVal col As Long, _
                    ByVal fontName As String, ByVal align As Long, ByVal anchor As Long)
    With s.TextFrame
        .WordWrap = msoTrue
        .MarginLeft = 14: .MarginRight = 14: .MarginTop = 6: .MarginBottom = 6
        .VerticalAnchor = anchor
        .TextRange.Text = t
        .TextRange.Font.Name = fontName
        .TextRange.Font.Size = size
        .TextRange.Font.Color.RGB = col
        .TextRange.ParagraphFormat.Alignment = align
    End With
End Sub

Private Function Pill(ByVal x As Single, ByVal y As Single, ByVal t As String, ByVal fillC As Long, _
                      ByVal textC As Long, Optional ByVal size As Single = 10, Optional ByVal caps As Boolean = False) As Shape
    Dim s As Shape
    Set s = Box(x, y, 60, 22, fillC, 0.5)
    With s.TextFrame
        .WordWrap = msoFalse
        .MarginLeft = 11: .MarginRight = 11: .MarginTop = 3: .MarginBottom = 3
        .VerticalAnchor = msoAnchorMiddle
        If caps Then .TextRange.Text = UCase(t) Else .TextRange.Text = t
        .TextRange.Font.Name = FONT_HEAD
        .TextRange.Font.Size = size
        .TextRange.Font.Color.RGB = textC
        .TextRange.ParagraphFormat.Alignment = ppAlignCenter
        .AutoSize = ppAutoSizeShapeToFitText
    End With
    If caps Then
        On Error Resume Next
        s.TextFrame2.TextRange.Font.Spacing = 1.2
        On Error GoTo 0
    End If
    s.Left = x
    s.Top = y
    Set Pill = s
End Function

' A rounded card with a bold heading paragraph and body text (optionally bulleted).
Private Function Card(ByVal x As Single, ByVal y As Single, ByVal w As Single, ByVal h As Single, _
                      ByVal fillC As Long, ByVal lineC As Long, ByVal head As String, ByVal body As String, _
                      ByVal headC As Long, ByVal bodyC As Long, ByVal headSize As Single, ByVal bodySize As Single, _
                      Optional ByVal bullets As Boolean = False) As Shape
    Dim s As Shape, i As Integer
    Set s = Box(x, y, w, h, fillC, 0.07, lineC)
    With s.TextFrame
        .WordWrap = msoTrue
        .AutoSize = ppAutoSizeNone
        .MarginLeft = 20: .MarginRight = 20: .MarginTop = 18: .MarginBottom = 16
        .VerticalAnchor = msoAnchorTop
        If body <> "" Then .TextRange.Text = head & vbCr & body Else .TextRange.Text = head
        .TextRange.ParagraphFormat.Alignment = ppAlignLeft
        .TextRange.ParagraphFormat.SpaceAfter = 5
        .TextRange.Font.Name = FONT_BODY
        .TextRange.Font.Size = bodySize
        .TextRange.Font.Color.RGB = bodyC
        With .TextRange.Paragraphs(1)
            .Font.Name = FONT_HEAD
            .Font.Size = headSize
            .Font.Color.RGB = headC
            .ParagraphFormat.SpaceAfter = 10
        End With
        If bullets And body <> "" Then
            On Error Resume Next
            For i = 2 To .TextRange.Paragraphs.Count
                With .TextRange.Paragraphs(i).ParagraphFormat.Bullet
                    .Visible = msoTrue
                    .Character = 8226
                    .UseTextColor = msoFalse
                    .Font.Color.RGB = headC
                End With
            Next i
            .Ruler.Levels(1).FirstMargin = 0
            .Ruler.Levels(1).LeftMargin = 16
            On Error GoTo 0
        End If
    End With
    Set Card = s
End Function

' Small caps label above a short paragraph.
Private Function LabelBody(ByVal x As Single, ByVal y As Single, ByVal w As Single, ByVal h As Single, _
                           ByVal label As String, ByVal body As String, ByVal labelC As Long) As Shape
    Dim s As Shape
    Set s = Txt(x, y, w, h, UCase(label) & vbCr & body, 13, cFg)
    StylePara s, 1, 9.5, labelC, FONT_HEAD, 1.2
    s.TextFrame.TextRange.Paragraphs(1).ParagraphFormat.SpaceAfter = 4
    s.TextFrame.TextRange.ParagraphFormat.SpaceWithin = 1.05
    Set LabelBody = s
End Function

' Soft navy shadow, used to lift a card off the page.
Private Sub Lift(ByVal s As Shape)
    On Error Resume Next
    With s.Shadow
        .Visible = msoTrue
        .ForeColor.RGB = cNavy
        .Transparency = 0.86
        .OffsetX = 0
        .OffsetY = 8
        .Blur = 22
    End With
    On Error GoTo 0
End Sub

' Yellow highlighter strokes behind each line of one paragraph, wiped in from the left.
Private Sub HighlightPara(ByVal t As Shape, ByVal paraIdx As Integer, ByVal startDelay As Single)
    Dim para As TextRange, ln As TextRange, hl As Shape, i As Integer
    On Error Resume Next
    Set para = t.TextFrame.TextRange.Paragraphs(paraIdx)
    For i = 1 To 4
        Set ln = Nothing
        Set ln = para.Lines(i, 1)
        If ln Is Nothing Then Exit For
        If Len(Trim(Replace(ln.Text, vbCr, ""))) = 0 Then Exit For
        Set hl = Box(ln.BoundLeft - 4, ln.BoundTop + ln.BoundHeight * 0.58, ln.BoundWidth + 8, ln.BoundHeight * 0.3, cYellow, 0.3)
        Anim hl, msoAnimEffectWipe, startDelay + (i - 1) * 0.3, 0.45, msoAnimDirectionLeft
    Next i
    t.ZOrder msoBringToFront
    On Error GoTo 0
End Sub

' Adds an entrance effect. The first effect on a slide starts after the transition;
' the rest start with it, offset by their delay, so the slide plays by itself.
Private Function Anim(ByVal s As Shape, ByVal eff As Long, ByVal delay As Single, ByVal dur As Single, _
                      Optional ByVal dirn As Long = 0) As Effect
    Dim e As Effect, trig As Long
    If firstAnim Then trig = msoAnimTriggerAfterPrevious Else trig = msoAnimTriggerWithPrevious
    On Error Resume Next
    Set e = sld.TimeLine.MainSequence.AddEffect(Shape:=s, effectId:=eff, trigger:=trig)
    If e Is Nothing Then
        Err.Clear
        Set e = sld.TimeLine.MainSequence.AddEffect(Shape:=s, effectId:=msoAnimEffectFade, trigger:=trig)
    End If
    If Not e Is Nothing Then
        e.Timing.Duration = dur
        e.Timing.TriggerDelayTime = delay
        If dirn <> 0 Then e.EffectParameters.Direction = dirn
    End If
    Err.Clear
    On Error GoTo 0
    firstAnim = False
    Set Anim = e
End Function

Private Function MinS(ByVal a As Single, ByVal b As Single) As Single
    If a < b Then MinS = a Else MinS = b
End Function

' Stylised app window: shell bar, side nav and four module tiles.
' Shapes are named with "!!" so Morph zooms the window between slides 1 and 2.
Private Sub AppWindow(ByVal x As Single, ByVal y As Single, ByVal w As Single, ByVal h As Single, _
                      ByVal labelled As Boolean, ByVal d As Single, ByVal animate As Boolean)
    Dim sx As Single, sy As Single, s As Shape, i As Integer, lc As Long
    Dim tx, ty, tf, nms, heads, subs
    sx = w / 400: sy = h / 280

    Set s = Box(x, y, w, h, cWhite, 0.06, cSkyStrong, "!!app_frame")
    Lift s
    If animate Then Anim s, msoAnimEffectFadedZoom, d, 0.7

    Set s = Box(x, y, w, 36 * sy, cNavy, MinS(0.5, 0.06 * MinS(w, h) / (36 * sy)), -1, "!!app_bar", msoShapeRound2SameRectangle)
    If labelled Then
        BoxText s, "App shell" & DOT & "owned by the platform team", 11, cWhite, FONT_HEAD, ppAlignLeft, msoAnchorMiddle
        s.TextFrame.MarginLeft = 64 * sx
    End If
    If animate Then Anim s, msoAnimEffectFade, d + 0.2, 0.4

    For i = 0 To 2
        Set s = Box(x + (16 + i * 14) * sx - 4, y + 18 * sy - 4, 8, 8, cOnNavy, 0, -1, "!!app_dot" & i, msoShapeOval)
        If animate Then Anim s, msoAnimEffectFade, d + 0.3, 0.4
    Next i

    Set s = Box(x + 16 * sx, y + 52 * sy, 72 * sx, 212 * sy, cSky, 0.08, -1, "!!app_nav")
    If labelled Then
        Dim nl As String
        nl = "Home" & vbCr & "Orders" & vbCr & "Reports" & vbCr & "Settings"
        BoxText s, nl, 10, cMuted, FONT_BODY, ppAlignLeft, msoAnchorTop
        s.TextFrame.MarginLeft = 10: s.TextFrame.MarginTop = 12
        s.TextFrame.TextRange.ParagraphFormat.SpaceAfter = 6
    End If
    If animate Then Anim s, msoAnimEffectFade, d + 0.3, 0.4

    tx = Array(100, 248, 100, 248)
    ty = Array(52, 52, 164, 164)
    tf = Array(cSky, cPink, cWhite, cYellow)
    nms = Array("!!app_tileA", "!!app_tileB", "!!app_tileC", "!!app_tileYou")
    heads = Array("Module A", "Module B", "Module C", "Your module")
    subs = Array("Team Alpha", "Team Beta", "Team Gamma", "Your team")
    For i = 0 To 3
        lc = -1
        If i = 2 Then lc = cLine
        If i = 3 Then lc = cNavy
        Set s = Box(x + tx(i) * sx, y + ty(i) * sy, 136 * sx, 100 * sy, tf(i), 0.1, lc, nms(i))
        If i = 3 Then s.Line.DashStyle = msoLineDash: s.Line.Weight = 1.75
        If labelled Then
            BoxText s, heads(i) & vbCr & subs(i), 10.5, cMuted, FONT_BODY, ppAlignLeft, msoAnchorTop
            s.TextFrame.MarginTop = 10: s.TextFrame.MarginLeft = 12
            StylePara s, 1, 13, cNavy, FONT_HEAD
        End If
        If animate Then Anim s, msoAnimEffectFadedZoom, d + 0.45 + i * 0.12, 0.5
    Next i
End Sub

        Array("App shell   /", "Platform", "[name]", "#platform-help"), _
        Array("[module]   /route", "[team]", "[name]", "#[channel]"), _
        Array("[module]   /route", "[team]", "[name]", "#[channel]"))

                .ForeColor.RGB = cSkyStrong
